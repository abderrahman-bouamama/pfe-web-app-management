## 1. installer composer `composer install`
## 2. Create .env file and personalise your database name example `LOG_CHANNEL=stack
LOG_STACK=single
LOG_DEPRECATIONS_CHANNEL=null
LOG_LEVEL=debug

DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=pfe_web_app_management
DB_USERNAME=root
DB_PASSWORD=`
## 3. run `php artisan migrate`
## 4. instal node modules , run `npm istall`
## 5. run `npm run dev`
## 6. run `php artisan serve`
